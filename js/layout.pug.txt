      document.addEventListener('DOMContentLoaded', function() {
        // --- 生成 HTML ---
        const switchLi = document.createElement('li');
        switchLi.id = 'sana-switch-li';
        switchLi.className = 'nav-item'; 
        switchLi.style.display = 'flex';
        switchLi.style.alignItems = 'center';

              <span class="sana-icon" id="sana-theme-icon">☀️</span>

            <div class="sana-switch-group" title="樱花/雪花特效">
              <span class="sana-icon" id="sana-effect-icon">🌸</span>
              <label class="ios-switch-mini">
                <input type="checkbox" id="sana-effect-toggle" checked>
                <span class="slider-mini"></span>

                        document.querySelector('nav ul');

        if (navMenu) {
          navMenu.insertBefore(switchLi, navMenu.firstChild);
        } else {
          switchLi.style.position = 'absolute';
          switchLi.style.top = '20px';
          switchLi.style.right = '100px';
          document.querySelector('header').appendChild(switchLi);
        }

        // --- 基础 UI 状态初始化 & 监听 ---
        const themeInput = document.getElementById('sana-theme-toggle');
        const effectInput = document.getElementById('sana-effect-toggle');
        const themeIcon = document.getElementById('sana-theme-icon');
        const effectIcon = document.getElementById('sana-effect-icon');

        function updateIcons(isDark) {
          themeIcon.textContent = isDark ? "🌙" : "☀️";
          effectIcon.textContent = isDark ? "❄️" : "🌸";
        }

        // 初始化开关状态
        const savedDarkMode = localStorage.getItem('darkMode') === 'true';
        themeInput.checked = savedDarkMode;
        updateIcons(savedDarkMode);
        
        const savedEffect = localStorage.getItem('sanaEffectEnabled');
        if (savedEffect !== null) effectInput.checked = (savedEffect === 'true');

        // 监听昼夜切换 (只负责切换 class，特效逻辑交给 weather.js)
        themeInput.addEventListener('change', function() {
          const isDark = themeInput.checked;
          updateIcons(isDark);
          if (isDark) {
            document.documentElement.classList.add('dark-mode');
            document.body.classList.add('dark-mode');
            localStorage.setItem('darkMode', 'true');
          } else {
            document.documentElement.classList.remove('dark-mode');
            document.body.classList.remove('dark-mode');
            localStorage.setItem('darkMode', 'false');
          }
        });
        
        // 特效开关只负责存状态，weather.js 会监听并处理
        effectInput.addEventListener('change', function() {
           // 这里的 update 交给 external script
           // 我们只需要更新 UI 或 LocalStorage
           // LocalStorage 更新已经在 weather.js 里处理了，这里留空或者双重保险
        });
      });
